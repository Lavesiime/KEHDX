// Events
// ========================

event ObjectUpdate
	if stage.activeLayer[0] == LAYER_INSIDE
		object.glowTimer++
		if object.glowTimer == 6
			object.glowTimer = 0
			
			object.glowFrame++
			object.glowFrame %= 9
			
			temp1 = object.glowFrame
			temp1 *= 5
			temp2 = 186
			while temp2 < 191
				GetTableValue(temp0, temp1, QQZSetup_rockPalette)
				SetPaletteEntry(0, temp2, temp0)
				
				temp1++
				temp2++
			loop
		end if
	end if
	
	object.convTimer++
	if object.convTimer == 5
		object.convTimer = 0
		
		object.convFrame++
		object.convFrame %= 3
		
		// (Waterfall palette cycle, uses the same timer as the conveyor belts)
		RotatePalette(0, 200, 202, true)
		
		temp1 = object.convFrame
		temp1 += object.direction
		temp1 *= 3
		temp2 = 170
		while temp2 < 173
			GetTableValue(temp0, temp1, QQZSetup_conveyorPalette)
			SetPaletteEntry(0, temp2, temp0)
			
			temp1++
			temp2++
		loop
	end if
	
	// Process Conveyor Belts
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if player[currentPlayer].gravity == GRAVITY_GROUND
			temp2 = player[currentPlayer].ypos
			temp2 >>= 16
			temp2 += player[currentPlayer].collisionBottom
			temp2 += 2
			
			temp1 = player[currentPlayer].xpos
			temp1 >>= 16
			temp1 += player[currentPlayer].collisionLeft
			
			Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_FLAGSA)
			temp3 = temp0
			
			temp1 = player[currentPlayer].xpos
			temp1 >>= 16
			temp1 += player[currentPlayer].collisionRight
			Get16x16TileInfo(temp0, temp1, temp2, TILEINFO_FLAGSA)
			temp3 |= temp0
			
			if temp3 == TILE_CONVEYORBELT
				if object.direction == CONVEYOR_RIGHT
					player[currentPlayer].xpos += object.speed
				else
					player[currentPlayer].xpos -= object.speed
				end if
			end if
		end if
	next
end event


event ObjectDraw
	currentPlayer = screen.currentID
end event

	object[SLOT_ZONESETUP].priority = PRIORITY_ACTIVE
	object[SLOT_ZONESETUP].drawOrder = 0
	object[SLOT_ZONESETUP].speed = 0x10000
end event
