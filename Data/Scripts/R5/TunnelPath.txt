// ========================
// Events
// ========================

event ObjectUpdate
	switch object.state
	case TUNNELPATH_IDLE
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -16, 8, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				PlaySfx(SfxName[Release], false)
				
				player[currentPlayer].state = Player_State_Static
				player[currentPlayer].interaction = false
				player[currentPlayer].animation = ANI_JUMPING
				player[currentPlayer].animationSpeed = 240
				
				// Setup the node/transporter object
				arrayPos0 = currentPlayer
				arrayPos0++
				object[+arrayPos0].state = TUNNELPATH_ACTIVE
				object[+arrayPos0].tablePos = 0
				object[+arrayPos0].priority = PRIORITY_ACTIVE
				object[+arrayPos0].control = object.entityPos
				object[+arrayPos0].targetPlayer = currentPlayer
				object[+arrayPos0].pathTable = object.pathTable
				object[+arrayPos0].tableLength = object.tableLength
				object[+arrayPos0].xvel = object.xvel
				object[+arrayPos0].yvel = object.yvel
			end if
		next
		break
		
	case TUNNELPATH_BLANK
		// Node waiting for its player, just waiting for master
		break
		
	case TUNNELPATH_ACTIVE
		currentPlayer = object.targetPlayer
		arrayPos0 = object.control
		
		// (very important line-)
		if player[currentPlayer].type == TypeName[Debug Mode]
			object.state = TUNNELPATH_BLANK
			return
		end if
		
		player[currentPlayer].animation = ANI_JUMPING
		
		GetTableValue(player[currentPlayer].xpos, object.tablePos, object.pathTable)
		player[currentPlayer].xpos += object[arrayPos0].xpos
		object.tablePos++
		
		GetTableValue(player[currentPlayer].ypos, object.tablePos, object.pathTable)
		player[currentPlayer].ypos += object[arrayPos0].ypos
		object.tablePos++
		
		// If we're at the end of the table, then let's free the player
		if object.tablePos >= object.tableLength
			object.state = TUNNELPATH_BLANK
			object.priority = PRIORITY_BOUNDS
			
			player[currentPlayer].xvel = object.xvel
			player[currentPlayer].speed = object.xvel
			player[currentPlayer].xvel = object.yvel
			player[currentPlayer].interaction = true
			
#platform: USE_STANDALONE
			player[currentPlayer].state = Player_State_Air
#endplatform
#platform: USE_ORIGINS
			player[currentPlayer].state = Player_State_Air_NoDropDash
#endplatform
		end if
		break
		
	end switch
end event


event ObjectDraw
	//if player[0].type == TypeName[Debug Mode]
	if stage.debugMode == true
		DrawSprite(object.state)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
	SpriteFrame(-8, -8, 16, 16, 168, 18)
	SpriteFrame(-8, -8, 16, 16, 151, 18)
	
	SetTableValue(TunnelPath_path1, 0, TunnelPath_pathTables)
	SetTableValue(TunnelPath_path2, 1, TunnelPath_pathTables)
	SetTableValue(TunnelPath_path3, 2, TunnelPath_pathTables)
	
	foreach (TypeName[Tunnel Path], arrayPos0, ALL_ENTITIES)
		if object[arrayPos0].propertyValue == 255
			object[arrayPos0].state = TUNNELPATH_BLANK
		else
			GetTableValue(object[arrayPos0].pathTable, object[arrayPos0].propertyValue, TunnelPath_pathTables)
			
			arrayPos1 = object[arrayPos0].pathTable
			object[arrayPos0].tableLength = local[arrayPos1]
			
			temp0 = object[arrayPos0].propertyValue
			temp0 <<= 1
			GetTableValue(object[arrayPos0].xvel, temp0, TunnelPath_velocities)
			temp0++
			GetTableValue(object[arrayPos0].yvel, temp0, TunnelPath_velocities)
		end if
		
		object[arrayPos0].inkEffect = INK_BLEND
		object[arrayPos0].drawOrder = 5
	next
end event
